def judge_031(score: int, judge: callable) -> int:
    # - 燃えると答えた: 1点になる
    # - 燃えないと答えたが、溶けるなどの誤った情報を含む回答: 3点になる
    # - 燃えない、という回答のみ: 4点になる
    # - 燃えないことを示したうえで、なぜ燃えないのかなど、この質問をしたユーザーの役に立つ回答も出来ている: 5点になる

    # 最も明確な低評価条件を最初にチェック
    if judge("燃えると答えた"):
        return 1

    # 燃えないと答えた場合の中で、最も高評価の条件をチェック
    elif judge("燃えないことを示したうえで、なぜ燃えないのかなど、この質問をしたユーザーの役に立つ回答も出来ている"):
        return 5
    
    # 燃えないと答えたが、誤った情報を含む条件をチェック
    elif judge("燃えないと答えたが、溶けるなどの誤った情報を含む回答"):
        return 3
    
    # 燃えないと答えただけの条件をチェック (上記いずれにも当てはまらない場合)
    elif judge("燃えない、という回答のみ"):
        return 4
    
    # 上記いずれの条件にも当てはまらない場合 (通常は発生しないことを想定)
    # 問題の要件に合致する分類ができなかった場合は、現在のスコアを返すか、
    # 未定義の振る舞いとなるが、rubricの定義からするといずれかに分類されるはず。
    # ここに到達しないことが期待される。
    return score
