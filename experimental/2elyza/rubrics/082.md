def judge_082(score: int, judge: callable) -> int:
    # - 言い換えの正誤判定に正解し、その上でユーザーの役に立つようにそれぞれの単語の意味の説明や、より適切な言い換えを提示する: 5点
    # - 言い換えの正誤判定に正解しただけ: 4点
    # - 正誤判定が不正解（言い換えが適切である）: 1点

    # 最も低い評価から優先的に判定
    # 正誤判定が不正解の場合、スコアは1点となる
    if judge("正誤判定が不正解である"):
        return 1
    
    # 正誤判定は正解しているが、それ以上の付加情報（単語の意味説明やより適切な言い換え）がない場合、スコアは4点となる
    # ここに到達している時点で、正誤判定は正解している
    if judge("ユーザーの役に立つようにそれぞれの単語の意味の説明や、より適切な言い換えを提示していない"):
        return 4
        
    # 上記のいずれの条件にも当てはまらない場合、
    # すなわち、正誤判定も正解しており、かつユーザーの役に立つ情報も提示している場合、スコアは5点（初期スコアのまま）となる
    return score
