def judge_041(score: int, judge: callable) -> int:
    # - 不正解の場合: 1点になる
    # - 情報不足、わからないといったように正解している: 4点になる
    # - 正解している上に理由を説明している: 5点になる

    if judge("正解している上に理由を説明している"):
        score = 5
    elif judge("情報不足、わからないといったように正解している"):
        score = 4
    elif judge("不正解である"):
        score = 1
    
    return score
