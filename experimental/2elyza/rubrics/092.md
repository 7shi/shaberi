def judge_092(score: int, judge: callable) -> int:
    # 情景を思い浮かび上がらせるような表現を追加して、風の強さをより強く表現している: 5点
    # 小説の文体としては不十分だが、風の強さをより強く表現している: 3点

    # まず、回答がどの「強い風の表現」の基準を満たしているかを判定し、ベースとなるスコアを設定する。
    # これは排他的に評価され、より高い基準を満たせばそのスコアが適用される。
    if judge("情景を思い浮かび上がらせるような表現を追加して、風の強さをより強く表現している"):
        score = 5
    elif judge("小説の文体としては不十分だが、風の強さをより強く表現している"):
        score = 3
    else:
        # どちらの「強い風の表現」の基準も満たさない場合、ベーススコアは0とする。
        # (初期スコアが5点として渡されても、この基準を満たさなければ低い点になるというELYZAの評価の解釈)
        score = 0

    # 減点項目:
    # - 弱い風の表現がある: -1点
    # - 不自然な表現: -1点
    
    if judge("弱い風の表現がある"):
        score -= 1
    if judge("不自然な表現がある"):
        score -= 1
        
    # スコアが0未満にならないようにする
    return max(0, score)
