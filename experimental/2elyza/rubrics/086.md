def judge_086(score: int, judge: callable) -> int:
    # 出題意図: ツールとその説明を与えたときにLLMがそれらをAgentのように使いこなせるかを見る
    # ベースとなる得点:
    # - ピザジャンボのSを注文した: 5点
    # - ラーメン屋に行った、ラーメンを出前として注文した: 3点
    # 減点項目:
    # - 捏造や誤りを含む: -1点

    # ベースとなる得点の判定
    # 最も良い結果（ピザジャンボSの注文）が達成されたかを確認
    if not judge("ピザジャンボのSを注文した"):
        # ピザジャンボSの注文が達成されていない場合、次の条件を確認
        if judge("ラーメン屋に行った、ラーメンを出前として注文した"):
            # ラーメンの注文が達成された場合、スコアを3点に設定
            score = 3
        else:
            # どちらの主要な行動も達成されていない場合、スコアを0点に設定
            score = 0
    # else: ピザジャンボSが注文された場合、scoreは初期値（通常5点）のまま

    # 減点項目の適用
    if judge("捏造や誤りを含む"):
        score -= 1

    # スコアが0点未満にならないように調整
    if score < 0:
        score = 0

    return score
