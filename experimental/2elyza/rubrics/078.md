def judge_078(score: int, judge: callable) -> int:
    # 評価基準
    # - Aに対応するQが書けている: 5点
    # - Qは書けているが、Aと対応していない: 3点
    # - Qが書けていない: 1点

    # 最も低いスコアの条件から順に判定し、スコアを上書きする
    if judge("Qが書けていない"):
        score = 1
    elif judge("Qは書けているが、Aと対応していない"):
        score = 3
    # Aに対応するQが書けている場合は、初期スコア（通常5点）がそのまま返される
    return score
