```python
def judge_061(score: int, judge: callable) -> int:
    # - 不正解の場合: 1点
    # - 正解しているが理由や説明がない場合: 4点
    # - 正解していて理由や説明がある場合: 5点

    if judge("不正解である"):
        score = 1
    elif judge("正解しているが理由や説明がない"):
        score = 4
    # 上記以外の場合（正解していて理由や説明がある場合）は、
    # 初期スコアが5点であればそのまま5点が維持される。
    
    return score
```
