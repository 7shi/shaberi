    # A. 何をやろうとしているか
    # B. なぜやろうとしているか
    # - A, Bを1つ間違えるごとに-2点
    
    # 「何をやろうとしているか」が間違っている、または記述がない場合に-2点
    if judge("「何をやろうとしているか」が間違っている、または記述がない"):
        score -= 2
    
    # 「なぜやろうとしているか」が間違っている、または記述がない場合に-2点
    if judge("「なぜやろうとしているか」が間違っている、または記述がない"):
        score -= 2
        
    return score
