def judge_076(score: int, judge: callable) -> int:
    # A. 全ての単語を使う
    # B. 順番に使う
    # C. 文として自然になっている
    # - A, B, Cそれぞれ間違えるごとに-2点

    if judge("全ての単語を使っていない"):
        score -= 2
    if judge("単語を順番通りに使っていない"):
        score -= 2

        score -= 2
