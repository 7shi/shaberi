def judge_081(score: int, judge: callable) -> int:
    # - 全問正解: 5点
    # - 1問のみ不正解: 4点
    # - 2問のみ不正解: 3点
    # - 3~4問不正解: 2点
    # - 5~6問不正解: 1点

    if judge("全問正解である"):
        return 5
    elif judge("1問のみ不正解である"):
        return 4
    elif judge("2問のみ不正解である"):
        return 3
    elif judge("3問または4問不正解である"):
        return 2
    elif judge("5問または6問不正解である"):
        return 1
    # 上記のいずれにも該当しない場合（例: 7問以上不正解など、評価基準に明示されていないケース）
    # 通常、このような評価基準では0点または最低点を返すことが多いが、
    # 提示された範囲をカバーすれば十分と判断し、それ以外のケースは想定しない。
    # もし、評価基準外のケースが存在し得るなら、別途定義が必要。
    # ここでは、提示された評価基準に厳密に従い、合致する条件でスコアを返す。
    # 仮にどれにも当てはまらないケースが入力された場合、初期スコアをそのまま返すのは不適切であるため、
    # 0点など最低点を返す設計も考えられるが、ELYZA-tasks-100の一般的なパターンとしては
    # 該当する条件を全て網羅しているため、elseに到達しない想定が多い。
    return 0 # どの条件にも合致しない場合（例：不正解が7問以上の場合など）は、最低点を返す
